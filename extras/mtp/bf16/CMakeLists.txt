cmake_minimum_required(VERSION 3.20)
project(bf16_mtp_reuse LANGUAGES C CXX)
set(ASCEND_CANN_PACKAGE_PATH /usr/local/Ascend/cann-9.2.0)
set(ENV{ASCEND_HOME_PATH} ${ASCEND_CANN_PACKAGE_PATH})
set(CMAKE_ASC_ARCHITECTURES dav-2201 CACHE STRING "architecture")
find_package(asc REQUIRED CONFIG PATHS "${ASCEND_CANN_PACKAGE_PATH}/compiler/tikcpp/ascendc_kernel_cmake" NO_DEFAULT_PATH)
enable_language(ASC)
foreach(group 2)
  ascendc_library(bf16_mtp_g${group} SHARED csrc/bf16_mtp.cpp)
  target_compile_definitions(bf16_mtp_g${group} PRIVATE QUERY_GROUP=${group})
  target_include_directories(bf16_mtp_g${group} PRIVATE "${ASCEND_CANN_PACKAGE_PATH}/include" "${ASCEND_CANN_PACKAGE_PATH}/compiler/tikcpp/tikcfw")
  target_link_directories(bf16_mtp_g${group} PRIVATE "${ASCEND_CANN_PACKAGE_PATH}/lib64")
  target_link_libraries(bf16_mtp_g${group} PRIVATE ascendcl runtime ascendc_runtime)
endforeach()
